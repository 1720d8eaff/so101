cmake_minimum_required(VERSION 3.16)
project(so101_control)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(trajectory_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(builtin_interfaces REQUIRED)
find_package(rosidl_default_generators REQUIRED)
rosidl_generate_interfaces(${PROJECT_NAME} "msg/ActionChunk.msg" DEPENDENCIES builtin_interfaces trajectory_msgs)
add_executable(control_node src/control_node.cpp)
target_compile_features(control_node PUBLIC cxx_std_17)
target_compile_options(control_node PRIVATE -Wall -Wextra -Werror)
target_include_directories(control_node PRIVATE include)
ament_target_dependencies(control_node rclcpp sensor_msgs trajectory_msgs std_msgs)
rosidl_get_typesupport_target(cpp_typesupport_target ${PROJECT_NAME} rosidl_typesupport_cpp)
target_link_libraries(control_node "${cpp_typesupport_target}")
install(TARGETS control_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
ament_export_dependencies(rosidl_default_runtime)
ament_package()
